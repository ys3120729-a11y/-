<script>
// 預設10年的報酬率與手動加碼歷史數據
const defaultData = [
    { year: 1, rate: 39, add: 36 },
    { year: 2, rate: 39, add: 0 },
    { year: 3, rate: -17, add: 20 },
    { year: 4, rate: 70, add: 0 },
    { year: 5, rate: 53.6, add: 0 },
    { year: 6, rate: 57.8, add: 0 },
    { year: 7, rate: -48, add: 40 },
    { year: 8, rate: 69.3, add: 0 },
    { year: 9, rate: 65.4, add: 0 },
    { year: 10, rate: 51.2, add: 0 }
];

// PMT 函數：計算每月還款額 (本息平均攤還)
function PMT(ratePerMonth, numberOfMonths, principal) {
    return (principal * ratePerMonth * Math.pow(1 + ratePerMonth, numberOfMonths)) / (Math.pow(1 + ratePerMonth, numberOfMonths) - 1);
}

function calculateModel() {
    // 讀取輸入值
    const initLoan = parseFloat(document.getElementById('initLoan').value) || 0;
    const loanRate = (parseFloat(document.getElementById('loanRate').value) || 0) / 100;
    const loanYears = parseInt(document.getElementById('loanYears').value) || 10;
    const pledgeRate = (parseFloat(document.getElementById('pledgeRate').value) || 0) / 100;

    // 計算信貸每月與每年結構
    const monthlyRate = loanRate / 12;
    const totalMonths = loanYears * 12;
    const monthlyPayment = PMT(monthlyRate, totalMonths, initLoan * 10000);
    const yearlyPaymentTotal = (monthlyPayment * 12) / 10000; // 萬元

    // 估算第一年還款結構
    let tempBalance = initLoan * 10000;
    let firstYearInterest = 0;
    for(let i=0; i<12; i++) {
        let interest = tempBalance * monthlyRate;
        let principal = monthlyPayment - interest;
        firstYearInterest += interest;
        tempBalance -= principal;
    }
    const firstYearInterestWan = firstYearInterest / 10000;
    const firstYearPrincipalWan = yearlyPaymentTotal - firstYearInterestWan;

    // 更新摘要文字
    document.getElementById('summaryText').innerHTML = `
        信貸每月還款額：<strong>${Math.round(monthlyPayment).toLocaleString()} 元</strong> 
        (一年共需自質押帳戶提領 <strong>${yearlyPaymentTotal.toFixed(2)} 萬</strong> 現金去還信貸本息)<br>
        每年信貸還款結構：還本金約 <strong>${firstYearPrincipalWan.toFixed(2)} 萬元</strong> ，純利息約 <strong>${firstYearInterestWan.toFixed(2)} 萬元</strong>。
    `;

    // 獲取目前表格中用戶最新數據
    const tbody = document.getElementById('tableBody');
    let currentInputs = [];
    if (tbody && tbody.children.length > 0) {
        for(let i=1; i<=10; i++) {
            const rateInput = document.getElementById(`rate_${i}`);
            const addInput = document.getElementById(`add_${i}`);
            currentInputs.push({
                rate: rateInput ? parseFloat(rateInput.value) : defaultData[i-1].rate,
                add: addInput ? parseFloat(addInput.value) : defaultData[i-1].add
            });
        }
    } else {
        currentInputs = defaultData.map(d => ({ rate: d.rate, add: d.add }));
    }

    // 清空並重新渲染表格
    tbody.innerHTML = '';

    // 年度 0 的初始狀態
    let marketValue = initLoan; 
    let debt = 0; 

    let html = `
        <tr>
            <td><strong>0</strong></td>
            <td>-</td>
            <td>${marketValue.toFixed(1)} 萬</td>
            <td>${initLoan.toFixed(1)} 萬</td>
            <td>-</td>
            <td>-</td>
            <td>-</td>
            <td>0.0 萬</td>
            <td>-</td>
            <td>-</td>
            <td>-</td>
            <td>${Math.max(0, (marketValue / 6 - debt)).toFixed(1)} 萬</td>
            <td>-</td>
            <td><span class="text-success">0.0 萬</span></td>
        </tr>
    `;

    // 遞迴計算 1~10 年
    for (let i = 1; i <= 10; i++) {
        const rowData = currentInputs[i-1];
        const rate = rowData.rate / 100;
        const manualAdd = rowData.add;

        // 1. 市場報酬變動
        let marketValueBeforeAdd = marketValue * (1 + rate);

        // 2. 質押利息 (若去年負債小於0，則利息以0計，不賺取銀行利息)
        let calcDebtForInterest = debt > 0 ? debt : 0;
        let currentPledgeInterest = calcDebtForInterest * pledgeRate;

        // 3. 現金流出
        let cashDrain = firstYearInterestWan; 

        // 4. 計算新負債
        debt = debt + currentPledgeInterest + cashDrain + manualAdd;

        // 5. 市值精準切入
        marketValue = marketValueBeforeAdd + manualAdd;

        // 6. 指標安全化修正
        let maintenanceRateStr = '-';
        let isLowMaintenance = false;
        let marginCallValue = 0;
        let safetyMargin = 0;

        if (debt > 0) {
            // 真正有欠債時計算精確數值
            let mRate = (marketValue / debt) * 100;
            maintenanceRateStr = Math.round(mRate) + '%';
            if (mRate < 140) isLowMaintenance = true;
            
            marginCallValue = debt * 1.3;
            if (marketValue > 0) {
                safetyMargin = ((marketValue - marginCallValue) / marketValue) * 100;
            }
        } else {
            // 負債小於等於0（代表倒賺現金），維持率是無限安全
            maintenanceRateStr = '∞ (無負債)';
            marginCallValue = 0;
            safetyMargin = 100; // 沒債務代表能承受100%下跌
        }

        // 600% 防線下最大可加碼額度
        let maxAdd600 = (marketValue / 6) - debt;
        if (maxAdd600 < 0) maxAdd600 = 0;

        // 純資產淨值
        let netWorth = marketValue - initLoan - debt;

        // 樣式標記
        let debtClass = debt > 0 ? 'class="text-danger"' : 'class="text-success"';
        let maintenanceClass = isLowMaintenance ? 'class="text-danger"' : 'class="text-success"';

        html += `
            <tr>
                <td><strong>${i}</strong></td>
                <td><input type="number" id="rate_${i}" class="table-input" value="${rowData.rate}" oninput="calculateModel()"> %</td>
                <td>${marketValue.toFixed(1)} 萬</td>
                <td>${initLoan.toFixed(1)} 萬</td>
                <td>${firstYearPrincipalWan.toFixed(1)} 萬</td>
                <td>${firstYearInterestWan.toFixed(1)} 萬</td>
                <td>${currentPledgeInterest.toFixed(1)} 萬</td>
                <td ${debtClass}>${debt.toFixed(1)} 萬</td>
                <td ${maintenanceClass}>${maintenanceRateStr}</td>
                <td>${marginCallValue > 0 ? marginCallValue.toFixed(1) + ' 萬' : '-'}</td>
                <td>${safetyMargin >= 100 ? '100%' : safetyMargin.toFixed(1) + '%'}</td>
                <td>${maxAdd600.toFixed(1)} 萬</td>
                <td><input type="number" id="add_${i}" class="table-input input-add" value="${manualAdd}" oninput="calculateModel()"></td>
                <td><strong class="text-success">${netWorth.toFixed(1)} 萬</strong></td>
            </tr>
        `;
    }

    if(tbody) tbody.innerHTML = html;
}

// 網頁載入時自動執行第一次計算
window.onload = calculateModel;
</script>
